% Separatrix system: Lorenz equations (Strogatz §9.3)
%   x' = sigma*(y - x)
%   y' = r*x - y - x*z
%   z' = x*y - b*z
% Parameters: sigma = 10, r = 28, b = 2.6666666666666665
% Needs: \usepackage{amsmath}
\begin{align}
  \dot{x} &= \sigma\left(y-x\right), \label{eq:lorenz-equations-x} \\
  \dot{y} &= rx-y-xz, \label{eq:lorenz-equations-y} \\
  \dot{z} &= xy-bz, \label{eq:lorenz-equations-z}
\end{align}
where $\sigma = 10$, $r = 28$ and $b = 2.66667$.
