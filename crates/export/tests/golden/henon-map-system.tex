% Separatrix system: Hénon map (Strogatz §12.2)
%   x_{n+1} = y_n + 1 - a*x_n^2
%   y_{n+1} = b*x_n
% Parameters: a = 1.4, b = 0.3
% Needs: \usepackage{amsmath}
\begin{align}
  x_{n+1} &= y_n+1-ax_n^2, \label{eq:henon-map-x} \\
  y_{n+1} &= bx_n, \label{eq:henon-map-y}
\end{align}
where $a = 1.4$ and $b = 0.3$.
